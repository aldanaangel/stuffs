#!/usr/bin/env bash

set -euo pipefail


# ============================================================
# PARAMETROS
# ============================================================

GROUP_ID="${1:?Debes indicar el GROUP_ID}"


# ============================================================
# CONFIGURACION
# ============================================================

GITLAB_URL="https://gitlab.midominio.com"

GITLAB_TOKEN="${GITLAB_TOKEN:?GITLAB_TOKEN no está definido}"

INCLUDE_SUBGROUPS="true"


# ============================================================
# VALIDAR DEPENDENCIAS
# ============================================================

command -v curl >/dev/null 2>&1 || {
    echo "[ERROR] curl no está instalado."
    exit 1
}

command -v jq >/dev/null 2>&1 || {
    echo "[ERROR] jq no está instalado."
    exit 1
}


# ============================================================
# VALIDAR SI EXISTE UNA RAMA
# ============================================================

branch_exists() {

    local project_id="$1"
    local branch="$2"

    local http_code

    http_code=$(
        curl --silent \
            --output /dev/null \
            --write-out "%{http_code}" \
            --header "PRIVATE-TOKEN: ${GITLAB_TOKEN}" \
            "${GITLAB_URL}/api/v4/projects/${project_id}/repository/branches/${branch}"
    )

    case "${http_code}" in

        200)
            return 0
            ;;

        404)
            return 1
            ;;

        *)
            echo "[ERROR] No fue posible consultar la rama '${branch}'." >&2
            echo "        Project ID: ${project_id}" >&2
            echo "        HTTP: ${http_code}" >&2
            return 2
            ;;
    esac
}


# ============================================================
# CREAR O ACTUALIZAR PROTECCION
# ============================================================

configure_branch_protection() {

    local project_id="$1"
    local project_name="$2"
    local branch="$3"
    local merge_level="$4"

    local http_code
    local temp_file

    echo
    echo "--------------------------------------------------"
    echo "Repo        : ${project_name}"
    echo "Project ID  : ${project_id}"
    echo "Branch      : ${branch}"
    echo "Push        : No one"
    echo "Merge level : ${merge_level}"
    echo "Force push  : Disabled"


    # ========================================================
    # VALIDAR QUE LA RAMA EXISTA
    # ========================================================

    if branch_exists "${project_id}" "${branch}"; then

        echo "[OK] Rama '${branch}' encontrada."

    else

        local status=$?

        if [[ "${status}" -eq 1 ]]; then

            echo "[SKIP] La rama '${branch}' no existe en '${project_name}'."
            return 0

        fi

        echo "[ERROR] No se pudo validar '${branch}'."
        return 1
    fi


    # ========================================================
    # CONSULTAR PROTECCION ACTUAL
    # ========================================================

    temp_file=$(mktemp)

    http_code=$(
        curl --silent \
            --show-error \
            --output "${temp_file}" \
            --write-out "%{http_code}" \
            --header "PRIVATE-TOKEN: ${GITLAB_TOKEN}" \
            "${GITLAB_URL}/api/v4/projects/${project_id}/protected_branches/${branch}"
    )


    # ========================================================
    # EXISTE -> ACTUALIZAR
    # ========================================================

    if [[ "${http_code}" == "200" ]]; then

        echo "[UPDATE] La rama ya está protegida."
        echo "         Analizando configuración actual..."


        # ----------------------------------------------------
        # CONTAR ACCESS LEVELS
        # ----------------------------------------------------

        local push_count
        local merge_count

        push_count=$(
            jq '.push_access_levels | length' "${temp_file}"
        )

        merge_count=$(
            jq '.merge_access_levels | length' "${temp_file}"
        )


        echo "         Push access levels : ${push_count}"
        echo "         Merge access levels: ${merge_count}"


        # ----------------------------------------------------
        # EVITAR MODIFICAR CONFIGURACIONES COMPLEJAS
        # ----------------------------------------------------

        if [[ "${push_count}" -gt 1 || "${merge_count}" -gt 1 ]]; then

            echo "[SKIP] La rama '${branch}' tiene múltiples access levels."
            echo "       No se modifica automáticamente por seguridad."
            echo
            echo "Configuración actual:"

            jq '{
                name,
                allow_force_push,
                push_access_levels,
                merge_access_levels
            }' "${temp_file}"

            rm -f "${temp_file}"

            return 0
        fi


        # ----------------------------------------------------
        # OBTENER IDs ACTUALES
        # ----------------------------------------------------

        local push_access_id
        local merge_access_id

        push_access_id=$(
            jq -r '.push_access_levels[0].id // empty' "${temp_file}"
        )

        merge_access_id=$(
            jq -r '.merge_access_levels[0].id // empty' "${temp_file}"
        )


        echo "         Push access ID : ${push_access_id:-NONE}"
        echo "         Merge access ID: ${merge_access_id:-NONE}"


        # ----------------------------------------------------
        # CONSTRUIR PAYLOAD
        #
        # Si existe access level -> incluir ID para modificarlo.
        # Si no existe -> no incluir ID para que GitLab lo cree.
        # ----------------------------------------------------

        local payload

        if [[ -n "${push_access_id}" && -n "${merge_access_id}" ]]; then

            payload=$(
                jq -n \
                    --argjson push_id "${push_access_id}" \
                    --argjson merge_id "${merge_access_id}" \
                    --argjson merge_level "${merge_level}" \
                    '{
                        allow_force_push: false,

                        allowed_to_push: [
                            {
                                id: $push_id,
                                access_level: 0
                            }
                        ],

                        allowed_to_merge: [
                            {
                                id: $merge_id,
                                access_level: $merge_level
                            }
                        ]
                    }'
            )

        elif [[ -z "${push_access_id}" && -n "${merge_access_id}" ]]; then

            payload=$(
                jq -n \
                    --argjson merge_id "${merge_access_id}" \
                    --argjson merge_level "${merge_level}" \
                    '{
                        allow_force_push: false,

                        allowed_to_push: [
                            {
                                access_level: 0
                            }
                        ],

                        allowed_to_merge: [
                            {
                                id: $merge_id,
                                access_level: $merge_level
                            }
                        ]
                    }'
            )

        elif [[ -n "${push_access_id}" && -z "${merge_access_id}" ]]; then

            payload=$(
                jq -n \
                    --argjson push_id "${push_access_id}" \
                    --argjson merge_level "${merge_level}" \
                    '{
                        allow_force_push: false,

                        allowed_to_push: [
                            {
                                id: $push_id,
                                access_level: 0
                            }
                        ],

                        allowed_to_merge: [
                            {
                                access_level: $merge_level
                            }
                        ]
                    }'
            )

        else

            payload=$(
                jq -n \
                    --argjson merge_level "${merge_level}" \
                    '{
                        allow_force_push: false,

                        allowed_to_push: [
                            {
                                access_level: 0
                            }
                        ],

                        allowed_to_merge: [
                            {
                                access_level: $merge_level
                            }
                        ]
                    }'
            )
        fi


        # ----------------------------------------------------
        # ACTUALIZAR
        # ----------------------------------------------------

        local update_response
        local update_http_code

        update_response=$(mktemp)

        update_http_code=$(
            curl --silent \
                --show-error \
                --output "${update_response}" \
                --write-out "%{http_code}" \
                --request PATCH \
                --header "PRIVATE-TOKEN: ${GITLAB_TOKEN}" \
                --header "Content-Type: application/json" \
                --data "${payload}" \
                "${GITLAB_URL}/api/v4/projects/${project_id}/protected_branches/${branch}"
        )


        if [[ "${update_http_code}" == "200" ]]; then

            echo "[OK] Protección actualizada."

            jq '{
                name,
                allow_force_push,
                push_access_levels,
                merge_access_levels
            }' "${update_response}"

        else

            echo "[ERROR] GitLab rechazó la actualización."
            echo "        Repo  : ${project_name}"
            echo "        Branch: ${branch}"
            echo "        HTTP  : ${update_http_code}"
            echo
            echo "Respuesta GitLab:"

            cat "${update_response}"
            echo

            rm -f "${update_response}"
            rm -f "${temp_file}"

            return 1
        fi

        rm -f "${update_response}"


    # ========================================================
    # NO EXISTE -> CREAR
    # ========================================================

    elif [[ "${http_code}" == "404" ]]; then

        echo "[CREATE] La rama no está protegida."
        echo "         Creando protección..."

        local payload

        payload=$(
            jq -n \
                --arg name "${branch}" \
                --argjson merge_level "${merge_level}" \
                '{
                    name: $name,
                    allow_force_push: false,

                    allowed_to_push: [
                        {
                            access_level: 0
                        }
                    ],

                    allowed_to_merge: [
                        {
                            access_level: $merge_level
                        }
                    ]
                }'
        )


        local create_response
        local create_http_code

        create_response=$(mktemp)

        create_http_code=$(
            curl --silent \
                --show-error \
                --output "${create_response}" \
                --write-out "%{http_code}" \
                --request POST \
                --header "PRIVATE-TOKEN: ${GITLAB_TOKEN}" \
                --header "Content-Type: application/json" \
                --data "${payload}" \
                "${GITLAB_URL}/api/v4/projects/${project_id}/protected_branches"
        )


        if [[ "${create_http_code}" == "201" ]]; then

            echo "[OK] Protección creada."

            jq '{
                name,
                allow_force_push,
                push_access_levels,
                merge_access_levels
            }' "${create_response}"

        else

            echo "[ERROR] GitLab rechazó la creación."
            echo "        Repo  : ${project_name}"
            echo "        Branch: ${branch}"
            echo "        HTTP  : ${create_http_code}"
            echo
            echo "Respuesta GitLab:"

            cat "${create_response}"
            echo

            rm -f "${create_response}"
            rm -f "${temp_file}"

            return 1
        fi

        rm -f "${create_response}"


    # ========================================================
    # ERROR CONSULTANDO PROTECCION
    # ========================================================

    else

        echo "[ERROR] No fue posible consultar la protección."
        echo "        Repo  : ${project_name}"
        echo "        Branch: ${branch}"
        echo "        HTTP  : ${http_code}"
        echo
        echo "Respuesta GitLab:"

        cat "${temp_file}"
        echo

        rm -f "${temp_file}"

        return 1
    fi


    rm -f "${temp_file}"
}


# ============================================================
# CONFIGURAR REPOSITORIO
# ============================================================

configure_project() {

    local project_id="$1"
    local project_name="$2"

    echo
    echo "=================================================="
    echo "CONFIGURANDO REPOSITORIO"
    echo "=================================================="
    echo "Repo : ${project_name}"
    echo "ID   : ${project_id}"
    echo "=================================================="


    # DEV
    # Push  : No one
    # Merge : Developers + Maintainers

    configure_branch_protection \
        "${project_id}" \
        "${project_name}" \
        "dev" \
        30


    # QA
    # Push  : No one
    # Merge : Developers + Maintainers

    configure_branch_protection \
        "${project_id}" \
        "${project_name}" \
        "qa" \
        30


    # MASTER
    # Push  : No one
    # Merge : Maintainers

    configure_branch_protection \
        "${project_id}" \
        "${project_name}" \
        "master" \
        40
}


# ============================================================
# INICIO
# ============================================================

echo
echo "=================================================="
echo "CONFIGURACION MASIVA DE PROTECTED BRANCHES"
echo "=================================================="
echo "Group ID: ${GROUP_ID}"
echo
echo "Politica:"
echo "  dev    -> Push: No one | Merge: Developers + Maintainers"
echo "  qa     -> Push: No one | Merge: Developers + Maintainers"
echo "  master -> Push: No one | Merge: Maintainers"
echo "  Force push -> Disabled"
echo "=================================================="


# ============================================================
# CONSULTAR PROYECTOS DEL GRUPO
# ============================================================

page=1

while true; do

    echo
    echo "Consultando proyectos - pagina ${page}..."

    response=$(
        curl --silent \
            --show-error \
            --fail-with-body \
            --header "PRIVATE-TOKEN: ${GITLAB_TOKEN}" \
            "${GITLAB_URL}/api/v4/groups/${GROUP_ID}/projects?include_subgroups=${INCLUDE_SUBGROUPS}&with_shared=false&per_page=100&page=${page}"
    )

    project_count=$(echo "${response}" | jq 'length')

    if [[ "${project_count}" -eq 0 ]]; then
        break
    fi


    # ========================================================
    # RECORRER REPOSITORIOS
    # ========================================================

    while IFS=$'\t' read -r project_id project_name; do

        configure_project \
            "${project_id}" \
            "${project_name}"

    done < <(
        echo "${response}" |
        jq -r '.[] | [.id, .path] | @tsv'
    )


    if [[ "${project_count}" -lt 100 ]]; then
        break
    fi

    page=$((page + 1))

done


echo
echo "=================================================="
echo "PROCESO TERMINADO"
echo "=================================================="
